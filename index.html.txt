    const canvas = document.getElementById('healthChart');
    const ctx = canvas.getContext('2d');

    };

    function drawChart() {
      const rect = chartWrap.getBoundingClientRect();
      const dpr = window.devicePixelRatio || 1;
      const width = rect.width;
      const height = rect.height;
      if (!width || !height) return;
      canvas.width = width * dpr;
      canvas.height = height * dpr;
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
      ctx.clearRect(0, 0, width, height);

      const data = rangeData[currentRange];
      let values = [...data.values];
      if (selectedAsset) {
        const adjust = selectedAsset.health - 81;
        values = values.map((v, i) => Math.max(28, Math.min(98, v + adjust + (i % 2 ? 1 : -1))));
      }
      const pad = { left: 35, right: 14, top: 13, bottom: 28 };
      const plotW = width - pad.left - pad.right;
      const plotH = height - pad.top - pad.bottom;
      const min = 40, max = 100;
      const y = val => pad.top + (max - val) / (max - min) * plotH;
      const x = i => pad.left + i * plotW / (values.length - 1);
      const gridVals = [40, 60, 80, 100];

      ctx.font = '10px DM Sans, sans-serif';
      gridVals.forEach(v => {
        const gy = y(v);
        ctx.beginPath();
        ctx.setLineDash([3, 4]);
        ctx.strokeStyle = '#edf1f3';
        ctx.lineWidth = 1;
        ctx.moveTo(pad.left, gy);
        ctx.lineTo(width - pad.right, gy);
        ctx.stroke();
        ctx.setLineDash([]);
        ctx.fillStyle = '#a0aab3';
        ctx.textAlign = 'right';
        ctx.textBaseline = 'middle';
        ctx.fillText(v, pad.left - 9, gy);
      });

      const thresholdY = y(60);
      ctx.beginPath();
      ctx.setLineDash([5, 4]);
      ctx.strokeStyle = '#e6aa4b';
      ctx.lineWidth = 1.2;
      ctx.moveTo(pad.left, thresholdY);
      ctx.lineTo(width - pad.right, thresholdY);
      ctx.stroke();
      ctx.setLineDash([]);

      const gradient = ctx.createLinearGradient(0, pad.top, 0, height - pad.bottom);
      gradient.addColorStop(0, 'rgba(19, 176, 132, .19)');
      gradient.addColorStop(1, 'rgba(19, 176, 132, 0)');
      ctx.beginPath();
      values.forEach((v, i) => i ? ctx.lineTo(x(i), y(v)) : ctx.moveTo(x(i), y(v)));
      ctx.lineTo(x(values.length - 1), height - pad.bottom);
      ctx.lineTo(x(0), height - pad.bottom);
      ctx.closePath();
      ctx.fillStyle = gradient;
      ctx.fill();

      ctx.beginPath();
      values.forEach((v, i) => i ? ctx.lineTo(x(i), y(v)) : ctx.moveTo(x(i), y(v)));
      ctx.strokeStyle = '#12a982';
      ctx.lineWidth = 2.5;
      ctx.lineJoin = 'round';
      ctx.lineCap = 'round';
      ctx.stroke();

      chartPoints = values.map((v, i) => ({ x: x(i), y: y(v), value: v, label: data.labels[i] }));
      chartPoints.forEach((p, i) => {
        ctx.beginPath();
        ctx.arc(p.x, p.y, i === values.length - 1 ? 4 : 3, 0, Math.PI * 2);
        ctx.fillStyle = '#fff';
        ctx.fill();
        ctx.lineWidth = 2;
        ctx.strokeStyle = '#12a982';
        ctx.stroke();
      });

      ctx.fillStyle = '#99a4ae';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'top';
      data.labels.forEach((label, i) => ctx.fillText(label, x(i), height - pad.bottom + 11));
      ctx.fillStyle = '#d69a3d';
      ctx.textAlign = 'left';
      ctx.fillText('Threshold', width - pad.right - 48, thresholdY - 13);
    }

    canvas.addEventListener('mousemove', event => {
      if (!chartPoints.length) return;
      const rect = canvas.getBoundingClientRect();
      const mx = event.clientX - rect.left;

    filterButton.addEventListener('click', () => {
      filterMode = (filterMode + 1) % 4;
      const modes = [
        { label: 'All assets', status: '' },
        { label: 'Optimal', status: 'optimal' },
        { label: 'Needs attention', status: 'watch' },
        { label: 'Critical', status: 'critical' }
      ];
      document.getElementById('filterLabel').textContent = modes[filterMode].label;
      rows.forEach(row => row.style.display = !modes[filterMode].status || row.dataset.status === modes[filterMode].status ? '' : 'none');
      document.getElementById('searchInput').value = '';
    });

    function showToast(message) {
      const toast = document.getElementById('toast');
      toast.textContent = message;
      toast.classList.add('show');
      clearTimeout(showToast.timer);
      showToast.timer = setTimeout(() => toast.classList.remove('show'), 2800);
    }

    document.getElementById('addAssetButton').addEventListener('click', () => {
      const name = prompt('Enter a name for your new asset:');
      if (!name || !name.trim()) return;
      const row = document.createElement('tr');
      row.dataset.name = name.trim();
      row.dataset.status = 'optimal';
      row.dataset.health = '100';
      row.innerHTML = `<td><div class="asset-cell"><span class="asset-icon"><svg viewBox="0 0 24 24" fill="none"><path d="M4 7.5 12 3l8 4.5v9L12 21l-8-4.5v-9Z" stroke="currentColor" stroke-width="1.7" stroke-linejoin="round"/><path d="m4.5 7.8 7.5 4.4 7.5-4.4M12 12.2V21" stroke="currentColor" stroke-width="1.7"/></svg></span><span class="asset-name"><strong>${escapeHTML(name.trim())}</strong><span>New asset · Not assigned</span></span></div></td><td><div class="health-mini"><span class="mini-track"><span style="width:100%"></span></span><b>100%</b></div></td><td><span class="table-status status-optimal">Connecting</span></td><td class="risk-low">Low</td><td>Just now</td>`;
      document.getElementById('assetRows').prepend(row);
      row.addEventListener('click', () => {
        rows.forEach(r => r.classList.remove('selected-row'));
        row.classList.add('selected-row');
        selectedAsset = { name: row.dataset.name, health: 100 };
        document.querySelector('.chart-panel .panel-subtitle').textContent = `${selectedAsset.name} · Health score trend`;
        drawChart();
      });
      document.getElementById('totalAssets').textContent = Number(document.getElementById('totalAssets').textContent) + 1;
      showToast(`${name.trim()} added to your asset list`);
    });

    function escapeHTML(str) {
      return str.replace(/[&<>"']/g, char => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[char]));
    }

    function refreshTelemetry() {
      const last = rangeData[currentRange].values;
      const latest = Math.max(50, Math.min(97, last[last.length - 1] + (Math.random() * 2 - 1)));
      rangeData[currentRange].values[rangeData[currentRange].values.length - 1] = Math.round(latest);
      document.getElementById('updatedTime').textContent = 'just now';
      if (!selectedAsset) drawChart();
    }
    setInterval(refreshTelemetry, 30000);
